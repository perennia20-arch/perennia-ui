        redis.disconnect();

async function upgradeVaultToWebCryptoAesGcm(
    page: Page,
    mnemonic: string,
    password: string
) {
    await page.evaluate(
        async ({ seedPhrase, pass }) => {
            const enc = new TextEncoder();
            const salt = window.crypto.getRandomValues(new Uint8Array(16));
            const iv = window.crypto.getRandomValues(new Uint8Array(12));
            const keyMaterial = await window.crypto.subtle.importKey(
                'raw',
                enc.encode(pass),
                { name: 'PBKDF2' },
                false,
                ['deriveKey']
            );
            const key = await window.crypto.subtle.deriveKey(
                {
                    name: 'PBKDF2',
                    salt,
                    iterations: 250000,
                    hash: 'SHA-256'
                },
                keyMaterial,
                { name: 'AES-GCM', length: 256 },
                false,
                ['encrypt', 'decrypt']
            );
            const encrypted = await window.crypto.subtle.encrypt(
                { name: 'AES-GCM', iv },
                key,
                enc.encode(seedPhrase)
            );
            const payload = new Uint8Array(
                salt.length + iv.length + encrypted.byteLength
            );
            payload.set(salt, 0);
            payload.set(iv, salt.length);
            payload.set(new Uint8Array(encrypted), salt.length + iv.length);
            let binary = '';
            for (let i = 0; i < payload.byteLength; i++) {
                binary += String.fromCharCode(payload[i]);
            }
            const existingRaw = localStorage.getItem('perennia_sovereign_payload');
            const parsed = existingRaw ? JSON.parse(existingRaw) : {};
            parsed.vault = window.btoa(binary);
            localStorage.setItem('perennia_sovereign_payload', JSON.stringify(parsed));
        },
        { seedPhrase: mnemonic, pass: password }
    );
}

                cleanInlineAccount.length < 4
            ) {
                globalFiatError =
                    'Please enter a complete 9-digit ABA Routing Number and Bank Account Number (minimum 4 digits).';
                return;
            }
            if (!inlineAbaTelemetry.checksumValid) {
                globalFiatError =
                    'ABA_CHECKSUM_INVALID: The 9-digit routing number failed the Federal Reserve Mod-10 checksum test.';
                return;
            }
        } else if (
            !savedBankLast4 &&
            !isBridgePushWire &&
            !isCardMethod &&
            globalFiatPaymentRail !== 'STRIPE_CRYPTO_ONRAMP'
        ) {
            showInlineBankSetup = true;
            globalFiatError =
                'Enter your 9-digit ABA Routing Number and Bank Account Number below to purchase via ACH, or select Card / Wire.';
            return;
        }
        isSubmittingGlobalFiat = true;
        globalFiatError = null;
        globalFiatSuccessOrder = null;
        const targetTicker = (ui.fiatModalTargetTicker || 'KAS').toUpperCase();
        const isAutoStakeSilo =
            targetTicker === DEFAULT_ECOSYSTEM_TICKER &&
            (ui.fiatModalAutoStakeSilo ||
                globalFiatDeliveryMode === 'AUTO_STAKE_SILO');
        const resolvedLabel =
            inlineBankLabel.trim() ||
            inlineAbaTelemetry.institutionName ||
            savedBankRailMeta?.bankLabel ||
            'Primary Operating Account';
        try {
            const payload: Record<string, any> = {
                walletAddress: identity.activeAddress,
                evmAddress: identity.sovereignKeys?.ethereum?.address,
                solAddress: identity.sovereignKeys?.solana?.address,
                fiatAmountUsd: usdAmt,
                targetAsset: targetTicker,
                paymentMethod: isBridgePushWire
                    ? 'wire'
                    : isCardMethod
                      ? 'card'
                      : 'ach',
                paymentRail: isBridgePushWire
                    ? 'BRIDGE_VIRTUAL_ACCOUNT_WIRE'
                    : isCardMethod
                      ? 'CARD_INSTANT'
                      : globalFiatPaymentRail,
                settlementTarget: isAutoStakeSilo
                    ? 'AUTO_STAKE_SILO'
                    : globalFiatDeliveryMode,
                autoStakeSilo: isAutoStakeSilo,
                bundledKasOrSilo: isAutoStakeSilo,
                instantBankDebit: !isBridgePushWire,
                bankLabel: resolvedLabel,
                accountType: inlineAccountType
            };
            if (
                cleanInlineRouting.length === 9 &&
                cleanInlineAccount.length >= 4
            ) {
                payload.routingNumber = cleanInlineRouting;
                payload.accountNumber = cleanInlineAccount;
            }
            const res = await fetch('/api/fiat/onramp', {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify(payload)
            });
            const data = await res.json().catch(() => ({}));
            if (!res.ok) {
                throw new Error(
                    data.error || 'Failed to execute bank purchase.'
                );
            }
            if (data.bankLast4) {
                savedBankLast4 = data.bankLast4;
                savedBankRailMeta = {
                    bankLabel: data.bankLabel || resolvedLabel,
                    accountType: inlineAccountType,
                    payoutMethod: 'ACH',
                    bankLast4: data.bankLast4,
                    updatedAt: new Date().toISOString()
                };
                inlineRoutingNumber = '';
                inlineAccountNumber = '';
                showInlineBankSetup = false;
            }
            globalFiatSuccessOrder = data;
            ui.triggerMicroBadge(
                'global_fiat_btn',
                data.status === 'COMPLETED'
                    ? `${targetTicker} PURCHASED & CREDITED âœ“`
                    : `ORDER ${data.orderId || data.sessionId} CREATED âœ“`,
                '#18C6A5'
            );
            if (identity.activeAddress) {
                await hydrateHeaderTreasuryBalances(identity.activeAddress);
            }
        } catch (err: any) {
            globalFiatError =
                err.message || 'Direct bank purchase execution failed.';
        } finally {
            isSubmittingGlobalFiat = false;
        }
    }

    async function settleGlobalFiatOrderNow() {
        if (
            !globalFiatSuccessOrder?.orderId &&
            !globalFiatSuccessOrder?.sessionId
        )
            return;
        const orderId =
            globalFiatSuccessOrder.orderId || globalFiatSuccessOrder.sessionId;
        isSettlingGlobalFiat = true;
        globalFiatError = null;
        try {
            const res = await fetch('/api/fiat/onramp', {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify({
                    action: 'SETTLE_ORDER_NOW',
                    orderId
                })
            });
            const data = await res.json().catch(() => ({}));
            if (!res.ok) {
                throw new Error(
                    data.error || 'Failed to settle purchase order.'
                );
            }
            globalFiatSuccessOrder = {
                ...globalFiatSuccessOrder,
                status: 'COMPLETED',
                txHash: data.txHash || globalFiatSuccessOrder.txHash,
                deliveredTokens:
                    data.deliveredTokens ??
                    globalFiatSuccessOrder.estimatedTokens
            };
            ui.triggerMicroBadge(
                'global_fiat_btn',
                `${globalFiatSuccessOrder.targetAsset || selectedGlobalTargetAsset.ticker} DELIVERED âœ“`,
                '#18C6A5'
            );
            if (identity.activeAddress) {
                await hydrateHeaderTreasuryBalances(identity.activeAddress);
            }
        } catch (err: any) {
            globalFiatError =
                err.message || 'Failed to complete order settlement.';
        } finally {
            isSettlingGlobalFiat = false;
        }
    }

    function hydratePersistedCustomTokens(customTokens: any[] | undefined) {
        if (!Array.isArray(customTokens)) return;
        for (const ct of customTokens) {
            const tick = String(ct?.ticker || ct?.symbol || '')
                .trim()
                .toUpperCase()
                .replace(/[^A-Z0-9]/g, '');
            if (!tick) continue;
            const existing = treasury.getAsset(tick);
            treasury.updateAsset(tick, {
                name: ct.name || existing.name || tick,
                assetClass: ct.assetClass || existing.assetClass || 'DeFi',
                network: (
                    ct.network ||
                    existing.network ||
                    'ETH'
                ).toUpperCase(),
                spotPrice:
                    Number(ct.spotPrice ?? ct.price ?? 0) > 0
                        ? Number(ct.spotPrice ?? ct.price)
                        : existing.spotPrice,
                delta:
                    ct.delta !== undefined || ct.change24h !== undefined
                        ? Number(ct.delta ?? ct.change24h ?? 0)
                        : existing.delta,
                contract: ct.contract || existing.contract || tick,
                imgUrl: ct.imgUrl || existing.imgUrl || '',
                theme: ct.theme ||
                    existing.theme || {
                        hex: '#18C6A5',
                        pastel: '#99f6e4'
                    }
            });
        }
    }

    async function hydrateHeaderTreasuryBalances(walletAddr: string) {
        if (!browser || !walletAddr) return;
        try {
            const requests: Promise<Response>[] = [
                fetch(`/api/balance?address=${encodeURIComponent(walletAddr)}`),
                fetch(
                    `/api/user/ledger?wallet=${encodeURIComponent(walletAddr)}`
                )
            ];
            if (identity.isCorporateAdmin && !ui.adminModeActive) {
                requests.push(
                    fetch(
                        `/api/treasury/corporate?corporate=true&kas=${encodeURIComponent(walletAddr)}`
                    )
                );
            }
            const [balRes, ledgerRes, corpRes] = await Promise.all(requests);
            if (balRes.ok) {
                const balData = await balRes.json();
                if (typeof balData.balance === 'number') {
                    treasury.updateAsset('KAS', { onChain: balData.balance });
                }
            }
            if (ledgerRes.ok) {
                const ledgerBalances = await ledgerRes.json();
                for (const [ticker, amount] of Object.entries(ledgerBalances)) {
                    if (
                        ticker !== 'unassignedReserveKas' &&
                        ticker !== 'unassignedReserveTargetKas'
                    ) {
                        treasury.updateAsset(ticker.toUpperCase(), {
                            synthetic: Number(amount) || 0
                        });
                    }
                }
            }
            if (corpRes && corpRes.ok) {
                const corpData = await corpRes.json();
                if (corpData?.reconciliation?.treasuryAccounting) {
                    adminTreasuryAccounting =
                        corpData.reconciliation.treasuryAccounting;
                }
            }
        } catch {
            // Ignore transient header balance refresh error
        }
    }

                if (typeof parsed.forgePublicEnabled === 'boolean') {
                    identity.forgePublicEnabled = parsed.forgePublicEnabled;

                    hydratePersistedCustomTokens(parsed.customTokens);

                    operations.updateEpochTelemetry(parsed.epochTelemetry);

    async function fetchPriceData() {
        try {
            const res = await fetch(
                '/api/prices?ids=kaspa,bitcoin,ethereum,solana,dogecoin,ripple,polygon-ecosystem-token,avalanche-2,sui,tron,zcash,arbitrum&include_24hr_change=true'
            );
            if (res.ok) {
                const data = await res.json();
                const map: Record<string, string> = {
                    kaspa: 'KAS',
                    bitcoin: 'BTC',
                    ethereum: 'ETH',
                    solana: 'SOL',
                    dogecoin: 'DOGE',
                    ripple: 'XRP',
                    'polygon-ecosystem-token': 'POL',
                    'avalanche-2': 'AVAX',
                    sui: 'SUI',
                    tron: 'TRX',
                    zcash: 'ZEC',
                    arbitrum: 'ARB'
                };
                const mappedKeys = new Set(Object.keys(map));
                for (const [cgKey, ticker] of Object.entries(map)) {
                    if (data?.[cgKey]?.usd !== undefined) {
                        const spotVal = Number(data[cgKey].usd);
                        treasury.updateAsset(ticker, {
                            spotPrice: spotVal,
                            delta: Number(data[cgKey].usd_24h_change || 0)
                        });
                        if (
                            ticker === 'KAS' &&
                            spotVal > 0 &&
                            treasury.getAsset(DEFAULT_ECOSYSTEM_TICKER)
                                .spotPrice <= 0
                        ) {
                            treasury.updateAsset(DEFAULT_ECOSYSTEM_TICKER, {
                                spotPrice: Number(
                                    (spotVal / 23.809524).toFixed(6)
                                )
                            });
                        }
                    }
                }
                for (const [rawKey, valObj] of Object.entries(data || {})) {
                    if (mappedKeys.has(rawKey)) continue;
                    const quote = valObj as any;
                    if (
                        quote &&
                        typeof quote.usd === 'number' &&
                        quote.usd > 0
                    ) {
                        const dynTicker = rawKey.trim().toUpperCase();
                        if (
                            treasury.registry.some(
                                (a) => a.ticker === dynTicker
                            )
                        ) {
                            treasury.updateAsset(dynTicker, {
                                spotPrice: Number(quote.usd),
                                delta: Number(quote.usd_24h_change || 0)
                            });
                        }
                    }
                }
            }
            if (identity.isWalletConnected && identity.activeAddress) {
                hydrateHeaderTreasuryBalances(
                    resolveTargetWallet() || identity.activeAddress
                );
            }
        } catch {
            // Ignore transient price fetch error
        }
    }

import { json, type RequestHandler, type RequestEvent } from '@sveltejs/kit';
import { dbPool } from '$lib/server/db';

    isWalletMasterAdmin,

    normalizeKaspaBase32Strict,
    perenniaKMS
} from '$lib/server/kms';

function resolveStratumTierEndpoint(hashrateTh: number): {
    tier: number;
    externalPort: number;
    internalPort: number;
    initialDifficulty: number;
    stratumUrl: string;
} {
    const host = (env.PUBLIC_STRATUM_HOST || 'pool.perennia.io').trim();
    if (hashrateTh > 10.0) {
        return {
            tier: 3,
            externalPort: 3334,
            internalPort: 6553,
            initialDifficulty: 1024.0,
            stratumUrl: `stratum+tcp://${host}:3334`
        };
    }
    if (hashrateTh >= 1.0) {
        return {
            tier: 2,
            externalPort: 3333,
            internalPort: 6552,
            initialDifficulty: 256.0,
            stratumUrl: `stratum+tcp://${host}:3333`
        };
    }
    return {
        tier: 1,
        externalPort: 3332,
        internalPort: 6551,
        initialDifficulty: 1.0,
        stratumUrl: `stratum+tcp://${host}:3332`
    };
}

async function ensureWorkerProfilesSchema() {
    await dbPool.query(`
        CREATE TABLE IF NOT EXISTS worker_profiles (
            worker_id VARCHAR(255) PRIMARY KEY,
            wallet_address VARCHAR(255) NOT NULL,
            worker_name VARCHAR(128) NOT NULL DEFAULT 'Rig01',
            hardware_model VARCHAR(128) NOT NULL DEFAULT 'IceRiver KS',
            current_hashrate DOUBLE PRECISION NOT NULL DEFAULT 0.0,
            valid_shares BIGINT NOT NULL DEFAULT 0,
            rejected_shares BIGINT NOT NULL DEFAULT 0,
            blocks_found BIGINT NOT NULL DEFAULT 0,
            stratum_port INTEGER NOT NULL DEFAULT 3332,
            last_seen TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
        );
    `);
}

async function hasDirectorClearance(
    request: Request,
    sessionWallet: string | null
): Promise<boolean> {
    const authHeader = request.headers.get('authorization') || '';
    if (authHeader.startsWith('Bearer ')) {
        try {
            if (perenniaKMS.verifyMasterAdminKey(authHeader)) {
                return true;
            }
        } catch {
            // Fall through to session check
        }
    }
    return isWalletMasterAdmin(sessionWallet);
}

export const GET: RequestHandler = async ({ cookies, url }: RequestEvent) => {
    const sessionId = cookies.get('perennia_session');
    const sessionWallet = sessionId
        ? await redis.get(`session:${sessionId}`).catch(() => null)
        : null;

    const requestedWallet = url.searchParams.get('wallet') || sessionWallet || '';
    const cleanWallet = normalizeKaspaBase32Strict(requestedWallet);

    const boundCoinbaseRaw =
        (await redis.get('perennia:stratum:mining_address').catch(() => null)) ||
        (env.PERENNIA_HOT_KAS_ADDRESS || '').trim() ||
        resolveMasterTreasuryKasAddressOptional() ||
        '';
    const poolCoinbaseAddress = boundCoinbaseRaw
        ? normalizeKaspaAddressStrict(boundCoinbaseRaw)
        : null;

    let workers: any[] = [];
    if (cleanWallet && sessionWallet) {
        const cleanSession = normalizeKaspaBase32Strict(sessionWallet);
        const isDirector = isWalletMasterAdmin(sessionWallet);
        if (cleanWallet === cleanSession || isDirector) {
            try {
                await ensureWorkerProfilesSchema();
                const res = await dbPool.query(
                    `SELECT worker_id, wallet_address, worker_name, hardware_model,
                            current_hashrate, valid_shares, rejected_shares,
                            blocks_found, stratum_port, last_seen
                     FROM worker_profiles
                     WHERE LOWER(REPLACE(TRIM(wallet_address), 'kaspa:', '')) = $1
                     ORDER BY last_seen DESC NULLS LAST`,
                    [cleanWallet]
                );
                workers = res.rows;
            } catch {
                workers = [];
            }
        }
    }

    return json({
        online: Boolean(poolCoinbaseAddress),
        poolCoinbaseAddress,
        tiers: [
            resolveStratumTierEndpoint(0.4),
            resolveStratumTierEndpoint(4.0),
            resolveStratumTierEndpoint(20.0)
        ],
        workers
    });
};

                })
            });

                throw new Error(
                    data.error || 'Failed to settle on-ramp order.'
                );
            }
            onrampSessionData = {
                ...onrampSessionData,
                status: 'COMPLETED',

            isSettlingOnrampPass = false;
        }
    }

    async function generateOfframpSettlementPass() {
        if (!identity.activeAddress) {
            ui.showWalletModal = true;
            return;
        }
        offrampError = null;
        offrampSuccessBanner = null;
        const grossUsd = parseFloat(offrampTargetUsdInput) || 0;
        const parsedCrypto =
            parseFloat(otcCryptoAmountInput) ||
            (offrampTargetSpotUsd > 0 ? grossUsd / offrampTargetSpotUsd : 0);
        if (grossUsd < 10 || parsedCrypto <= 0) {
            offrampError = 'Minimum sale amount is $10.00 USD.';
            return;
        }
        if (
            !savedBankLast4 &&
            (!abaTelemetry.checksumValid ||
                offrampAccountNumber.replace(/\D/g, '').length < 4)
        ) {
            offrampError =
                'Link or enter your 9-digit routing number and account number first.';
            return;
        }
        const isLedgerSale =
            offrampSourceMode === 'platform_balance' ||
            offrampSourceMode === 'ledger_kas' ||
            offrampSourceMode === 'ledger_usdc';
        if (isLedgerSale && parsedCrypto > offrampAvailableSourceBalance + 1e-8) {
            offrampError = `Insufficient ${offrampAssetTicker} in Platform Balance (${offrampAvailableSourceBalance.toFixed(4)} available).`;
            return;
        }
        try {
            const resolvedSourceMode = isLedgerSale
                ? offrampAssetTicker === 'USDC'
                    ? 'ledger_usdc'
                    : 'platform_balance'
                : 'direct_wholesale_deposit';
            const extBasisParsed = parseFloat(externalCostBasisUsdInput);
            const payload: Record<string, any> = {
                wallet: identity.activeAddress,
                amountKas: parsedCrypto,
                kasAmount: parsedCrypto,
                amountAsset: parsedCrypto,
                grossUsd,
                sourceMode: resolvedSourceMode,
                sourceAsset: offrampAssetTicker,
                payoutMethod,
                bankLabel:
                    offrampBankLabel.trim() || 'Primary Operating Account',
                accountType: offrampAccountType,
                ...(Number.isFinite(extBasisParsed) && extBasisParsed >= 0
                    ? { externalCostBasisUsd: extBasisParsed }
                    : {})
            };
            if (offrampRoutingNumber.trim() && offrampAccountNumber.trim()) {
                payload.routingNumber = offrampRoutingNumber.trim();
                payload.accountNumber = offrampAccountNumber.trim();
            }
            const res = await fetch('/api/fiat/offramp', {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify(payload)
            });
            const data = await res.json().catch(() => ({}));
            if (!res.ok) {
                offrampError = data.error || 'Off-Ramp order rejected.';
                return;
            }
            activeOfframpPass = {
                ...data,
                payoutId: data.payoutId || data.wireReferenceId,
                sourceAsset: data.sourceAsset || offrampAssetTicker,
                amountCrypto: data.amountCrypto ?? parsedCrypto,
                grossUsd: data.grossUsd ?? offrampCostPlusQuote.grossUsd,
                feeUsd: data.feeUsd ?? offrampCostPlusQuote.feeUsd,
                hardCostUsd:
                    data.hardCostUsd ?? offrampCostPlusQuote.hardCostUsd,
                guaranteedProfitUsd:
                    data.guaranteedProfitUsd ??
                    offrampCostPlusQuote.guaranteedProfitUsd,
                guestSavingsUsd:
                    data.guestSavingsUsd ??
                    offrampCostPlusQuote.guestSavingsUsd,
                netFiatUsd:
                    data.netFiatPayout ??
                    data.netFiatUsd ??
                    offrampCostPlusQuote.guestNetUsd,
                depositAddress:
                    data.depositAddress ||
                    offrampDepositPreview.address ||
                    `bc1qperenniawholesale${offrampAssetTicker.toLowerCase()}`,
                depositRouteType:
                    data.depositRouteType || offrampDepositPreview.routeLabel,
                confirmationsRequired:
                    data.confirmationsRequired ||
                    offrampDepositPreview.confirmationsRequired,
                confirmationsCurrent: data.confirmationsCurrent ?? 0,
                status: data.status || 'AWAITING_ONCHAIN_CONFIRMATIONS'
            };
            activeStep = 3;
            offrampSuccessBanner = isLedgerSale
                ? `Sold ${parsedCrypto.toFixed(4)} ${offrampAssetTicker} â†’ $${Number(activeOfframpPass.netFiatUsd).toFixed(2)} USD Queued (${activeOfframpPass.payoutId})`

            ui.triggerMicroBadge(
                'dex_fiat_btn',
                `OFF-RAMP ${activeOfframpPass.payoutId} ARMED âœ“`,
                '#18C6A5'

            await onSyncUserBalances();
            await checkFiatGatewayStatus();
        } catch (err: any) {
            offrampError = err.message || 'Unable to generate Off-Ramp order.';
        }
    }

    async function simulateOfframpConfirmNow() {
        if (!activeOfframpPass?.payoutId) return;
        activeOfframpPass = {
            ...activeOfframpPass,
            confirmationsCurrent:
                activeOfframpPass.confirmationsRequired || 6,
            status: 'DISPATCHED_BY_DIRECTOR',
            traceId: `ACH-TRACE-${Date.now().toString(36).toUpperCase()}`
        };
        offrampSuccessBanner = `Bank Payout Dispatched (${activeOfframpPass.payoutId}) âœ“`;
        await onSyncUserBalances();
    }

    function signAndBroadcastOfframpFromWallet() {
        if (!activeOfframpPass) return;
        const dest =
            activeOfframpPass.depositAddress || masterTreasuryKasAddress;
        if (!dest) {
            offrampError = 'Deposit address unavailable for wallet signing.';
            return;
        }
        txState.lastTxId = '';
        txState.lastExplorerUrl = '';
        awaitingWalletOfframpSign = true;
        requestUniversalL1Signature({
            intentType: 'FIAT_OFFRAMP',
            payAsset: activeOfframpPass.sourceAsset || offrampAssetTicker,
            payAmount:
                activeOfframpPass.amountCrypto || otcCryptoAmountInput || '0',
            fiatAmountUsd: Number(activeOfframpPass.netFiatUsd || 0).toFixed(2),
            destinationAddress: dest,
            payoutMethod,
            bankLabel: offrampBankLabel,
            accountType: offrampAccountType,
            orderId: activeOfframpPass.payoutId
        });
    }

    async function pollActiveOnrampSession() {
        if (!onrampSessionData?.sessionId) return;
        try {
            const res = await fetch(
                `/api/fiat/onramp?sessionId=${encodeURIComponent(onrampSessionData.sessionId)}`
            );
            if (res.ok) {
                const data = await res.json();
                if (data.session) {
                    onrampSessionData = {
                        ...onrampSessionData,
                        status: data.session.status || onrampSessionData.status,
                        txHash: data.session.txHash || onrampSessionData.txHash
                    };
                    if (data.session.status === 'COMPLETED') {
                        onSyncUserBalances();
                        checkFiatGatewayStatus();
                    }
                }
            }
        } catch {
            // Ignore transient poll error
        }
    }

    function handleLiveFiatSseEvent(e: Event) {
        const customEvt = e as CustomEvent;
        const payload = customEvt.detail;
        if (!payload) return;
        const myClean = (identity.activeAddress || '')
            .toLowerCase()
            .replace(/^kaspa:/i, '')
            .trim();
        const evtClean = String(payload.wallet || '')
            .toLowerCase()
            .replace(/^kaspa:/i, '')
            .trim();
        if (myClean && evtClean === myClean) {
            if (payload.eventType === 'FIAT_ONRAMP') {
                if (
                    onrampSessionData &&
                    payload.sessionId === onrampSessionData.sessionId
                ) {
                    onrampSessionData = {
                        ...onrampSessionData,
                        status: payload.status || 'COMPLETED',
                        txHash: payload.txHash || onrampSessionData.txHash
                    };
                }
            } else if (payload.eventType === 'FIAT_DISPATCHED') {
                offrampSuccessBanner = `Bank Transfer Dispatched (${payload.payoutId} â€¢ Trace: ${payload.traceId || 'CONFIRMED'})`;
                if (
                    activeOfframpPass &&
                    activeOfframpPass.payoutId === payload.payoutId
                ) {
                    activeOfframpPass = {
                        ...activeOfframpPass,
                        status: 'DISPATCHED_BY_DIRECTOR',
                        traceId: payload.traceId
                    };
                }
            }
            onSyncUserBalances();
            checkFiatGatewayStatus();
        }
    }

    onMount(() => {
        checkFiatGatewayStatus();
        if (browser) {
            window.addEventListener(
                'perennia:fiat_event',
                handleLiveFiatSseEvent
            );
        }
        onrampPollInterval = setInterval(() => {
            if (
                onrampSessionData?.sessionId &&
                onrampSessionData.status !== 'COMPLETED'
            ) {
                pollActiveOnrampSession();
            }
        }, 6000);
    });

    onDestroy(() => {
        clearTimeout(discoverTimer);
        if (browser) {
            window.removeEventListener(
                'perennia:fiat_event',
                handleLiveFiatSseEvent
            );
        }
        if (onrampPollInterval) clearInterval(onrampPollInterval);
    });
</script>

// src/lib/server/sor.ts

import { env } from '$env/dynamic/private';

export interface SwapPayload {
    payAsset: string;
    receiveAsset: string;
    amount: number;
}

export interface WaterfallLeg {
    tier: number;
    provider: string;
    filledAmount: number;
    executionRate: number;
    marginCaptured: number;
}

export interface SorExecutionPlan {
    totalAmount: number;
    unifiedRate: number;
    legs: WaterfallLeg[];
    estimatedOutput: number;
    transaction_uuid?: string;
    psbt?: any;
}

export class SmartOrderRouter {
    public async calculateSplitFill(req: SwapPayload, wallet: string, utxos?: any[], systemMode: string = "base"): Promise<SorExecutionPlan> {
        
        if (!env.RUST_BACKEND_URL) {
            throw new Error("CRITICAL: RUST_BACKEND_URL is missing. L1 Routing Unavailable.");
        }

        try {
            const res = await fetch(`${env.RUST_BACKEND_URL}/v1/sor/execute`, {
                method: 'POST',
                headers: {
                    'Content-Type': 'application/json',
                    'Authorization': `Bearer ${env.INTERNAL_ADMIN_TOKEN || env.PERENNIA_ADMIN_KEY || ''}`
                },
                body: JSON.stringify({
                    wallet,
                    payAsset: req.payAsset,
                    receiveAsset: req.receiveAsset,
                    amount: req.amount,
                    slippageTolerance: 0.05,
                    utxos: utxos || null,
                    systemMode: systemMode
                })
            });

            if (!res.ok) {
                const err = await res.json().catch(() => ({}));
                throw new Error(`Rust SOR Engine Rejected Payload: ${err.error || 'Unknown Error'}`);
            }

            return await res.json();
        } catch (e: any) {
            console.error(`ðŸš¨ FATAL: Rust SOR Engine Unreachable/Failed: ${e.message}`);
            throw new Error(`L1 Routing Unavailable: ${e.message}`);
        }
    }
}

export const sorEngine = new SmartOrderRouter();
